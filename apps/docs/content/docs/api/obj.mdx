# preludejs/Obj API Reference

## Core

### clone <span className="inline-flex items-center rounded-full border border-emerald-200 dark:border-emerald-800 bg-emerald-50 dark:bg-emerald-950/30 text-emerald-600 dark:text-emerald-400 text-xs font-semibold px-2.5 py-0.5 font-medium ml-2 align-middle">ƒ</span>

Returns a deep clone of the value via JSON round-tripping.

```typescript
function clone<T extends object>(obj: T): T
```

#### Parameters

| Parameter | Type | Description |
|---|---|---|
| `obj` | `T` |  |

#### Example

```ts
clone({ a: 1, b: 2 }) //=> { a: 1, b: 2 }
```

### get <span className="inline-flex items-center rounded-full border border-emerald-200 dark:border-emerald-800 bg-emerald-50 dark:bg-emerald-950/30 text-emerald-600 dark:text-emerald-400 text-xs font-semibold px-2.5 py-0.5 font-medium ml-2 align-middle">ƒ</span>

Reads a member from an object by key; key-first and curried.

```typescript
function get<T extends Record<PropertyKey, unknown>, K extends keyof T>(key: K): (obj: T) => T[K]
function get<T extends Record<PropertyKey, unknown>, K extends keyof T>(key: K, obj: T): T[K]
```

#### Parameters

| Parameter | Type | Description |
|---|---|---|
| `key` | `K` |  |

#### Example

```ts
get('foo', { foo: 'bar' }) //=> 'bar'
```

### map <span className="inline-flex items-center rounded-full border border-emerald-200 dark:border-emerald-800 bg-emerald-50 dark:bg-emerald-950/30 text-emerald-600 dark:text-emerald-400 text-xs font-semibold px-2.5 py-0.5 font-medium ml-2 align-middle">ƒ</span>

Maps each key/value pair of an object to an array; curried.

```typescript
function map<V, B>(fn: (key: string, value: V, index: number) => B): (obj: Record<string, V>) => B[]
function map<V, B>(fn: (key: string, value: V, index: number) => B, obj: Record<string, V>): B[]
```

#### Parameters

| Parameter | Type | Description |
|---|---|---|
| `fn` | `(key: string, value: V, index: number) => B` |  |

#### Example

```ts
map((k: string, v: number) => `${k}=${v}`, { a: 1, b: 2 }) //=> ['a=1', 'b=2']
```

<hr className="my-8 border-neutral-200 dark:border-neutral-800" />

### merge <span className="inline-flex items-center rounded-full border border-emerald-200 dark:border-emerald-800 bg-emerald-50 dark:bg-emerald-950/30 text-emerald-600 dark:text-emerald-400 text-xs font-semibold px-2.5 py-0.5 font-medium ml-2 align-middle">ƒ</span>

Merges any number of source objects into the first; later sources win.

```typescript
function merge<T extends object>(target: T, ...sources: Partial<T>[]): T
```

#### Parameters

| Parameter | Type | Description |
|---|---|---|
| `target` | `T` |  |
| `sources` | `Partial<T>[]` |  |

#### Example

```ts
merge({ a: 1, b: 2 }, { b: 3 }) //=> { a: 1, b: 3 }
```

```typescript
function objToPairs<T extends Record<PropertyKey, unknown>>(obj: T): (readonly [keyof T, T[keyof T]])[]
```

#### Parameters

| Parameter | Type | Description |
|---|---|---|
| `obj` | `T` |  |

```ts
objToPairs({ a: 'b', c: 'd' }) //=> [['a', 'b'], ['c', 'd']]
```

### pairsToObj <span className="inline-flex items-center rounded-full border border-emerald-200 dark:border-emerald-800 bg-emerald-50 dark:bg-emerald-950/30 text-emerald-600 dark:text-emerald-400 text-xs font-semibold px-2.5 py-0.5 font-medium ml-2 align-middle">ƒ</span>

Turns a list of `[key, value]` pairs into an object.

```typescript
function pairsToObj<K extends PropertyKey, V>(pairs: readonly (readonly [K, V])[]): Record<K, V>
```

#### Parameters

| Parameter | Type | Description |
|---|---|---|
| `pairs` | `readonly (readonly [K, V])[]` |  |

#### Example


### reduce <span className="inline-flex items-center rounded-full border border-emerald-200 dark:border-emerald-800 bg-emerald-50 dark:bg-emerald-950/30 text-emerald-600 dark:text-emerald-400 text-xs font-semibold px-2.5 py-0.5 font-medium ml-2 align-middle">ƒ</span>

Reduces an object's key/value pairs to a single value; curried.
The callback receives `(acc, key, value, index, obj)`.

```typescript
function reduce<A, V>(fn: (acc: A, key: string, value: V, index: number, obj: Record<string, V>) => A): (initial: A) => (obj: Record<string, V>) => A
function reduce<A, V>(fn: (acc: A, key: string, value: V, index: number, obj: Record<string, V>) => A, initial: A): (obj: Record<string, V>) => A
function reduce<A, V>(fn: (acc: A, key: string, value: V, index: number, obj: Record<string, V>) => A, initial: A, obj: Record<string, V>): A
```

#### Parameters

| Parameter | Type | Description |
|---|---|---|
| `fn` | `(acc: A, key: string, value: V, index: number, obj: Record<string, V>) => A` |  |

#### Example

```ts
reduce((acc: number, _k: string, v: number) => acc + v, 0, { a: 1, b: 2 }) //=> 3
```
